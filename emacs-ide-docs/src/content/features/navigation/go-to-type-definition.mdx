commands:
  - action: "Jump to type definition"
    cmd: "eglot-find-typeDefinition"
    key: "<kbd>C-c t</kbd> / <kbd>g t</kbd>"
    notes: "Jumps to the type (class/interface/struct) of the symbol\n                        at point."
  - action: "Jump to definition (contrast)"
    cmd: "xref-find-definitions"
    key: "<kbd>M-.</kbd> / <kbd>g d</kbd>"
    notes: "Jumps to the value/implementation of the symbol."
  - action: "Jump to declaration (contrast)"
    cmd: "eglot-find-declaration"
    key: "<kbd>C-c d</kbd> / <kbd>g D</kbd>"
    notes: "Jumps to the header/forward declaration."
  - action: "Go back (history)"
    cmd: "xref-go-back"
    key: "<kbd>M-,</kbd> / <kbd>g ,</kbd>"
    notes: "Returns to the exact cursor position before the jump."
  - action: "Peek type definition (inline)"
    cmd: "peek-xref-definition"
    key: "<kbd>SPC c p d</kbd>"
    notes: "Requirespeekpackage; renders the type\n                        inline without switching tabs."
parity_matrix:
  - vscode: "Right-click → Go to Type Definition"

  - vscode: "Dropdown for ambiguous types"

  - vscode: "Fallback to definition if type missing"
    emacs: "<code>eglot</code> natively falls back to definition if\n                        the LSP server returns an empty type definition payload."
