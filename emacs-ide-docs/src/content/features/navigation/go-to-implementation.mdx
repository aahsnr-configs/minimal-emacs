    notes: "Jumps to the interface/abstract definition itself."
  - action: "Go back (history)"
    cmd: "xref-go-back"
    key: "<kbd>M-,</kbd> / <kbd>g ,</kbd>"
    notes: "Returns to the exact cursor position before the jump."
  - action: "Go forward (history)"
    cmd: "xref-go-forward"
    key: "—"
    notes: "Reversesxref-go-back."
  - action: "Peek implementation (inline)"
    cmd: "peek-xref-definition"
    key: "<kbd>SPC c p d</kbd>"
    notes: "Requirespeekpackage; renders the target\n                        inline without switching tabs."
