** DONE Vertico
Renders completion candidates in a vertical, interactive dropdown list.
Core engine only — bundled extensions (directory, quick, repeat, multiform)
are loaded as separate =use-package= blocks per the official documentation
pattern, guaranteeing correct load order under Elpaca's async queue.
#+begin_src emacs-lisp
(use-package vertico
  :demand t
  :custom
  ;; Reserves exactly 10 lines of vertical space for the dropdown.
  (vertico-count 10)
  ;; Disables dynamic resizing to prevent UI jumping.
  (vertico-resize nil)
  ;; Allows cycling from the bottom of the list back to the top.
  (vertico-cycle t)
  :bind (:map vertico-map
    ;; Group navigation via arrow keys.
    ("C-<down>" . vertico-next-group)
    ("C-<up>"   . vertico-previous-group))
  :config
  ;; Prevents cursor wandering behind the read-only prompt text.
  (setq minibuffer-prompt-properties
        '(read-only t cursor-intangible t face minibuffer-prompt))
  (add-hook 'minibuffer-setup-hook #'cursor-intangible-mode)
  ;; NOTE: `enable-recursive-minibuffers' is set in the savehist section
  ;; (single source of truth). Do not duplicate here.
  ;; Displays depth indicator for nested recursive minibuffers.
  (minibuffer-depth-indicate-mode 1)
  ;; Activates the core Vertico engine post-load.
  (vertico-mode 1))
#+end_src

** DONE Vertico Directory
Ido-like directory navigation commands for the minibuffer. Bundled extension;
loaded separately per official documentation to guarantee autoload resolution
under Elpaca.
#+begin_src emacs-lisp
(use-package vertico-directory
  :ensure nil
  :after vertico
  :bind (:map vertico-map
    ("RET"   . vertico-directory-enter)
    ("DEL"   . vertico-directory-delete-char)
    ("M-DEL" . vertico-directory-delete-word))
  :hook (rfn-eshadow-update-overlay . vertico-directory-tidy))
#+end_src

** DONE Vertico Multiform
Per-command and per-category display configuration. Bundled extension;
loaded separately per official documentation. Houses the candidate
transform engine (wiki pattern) for directory and enabled-mode highlighting.
#+begin_src emacs-lisp
(use-package vertico-multiform
  :ensure nil
  :after vertico
  :config
  ;; ──────────────────────────────────────────────────────────────
  ;; Candidate Transform Engine (Vertico wiki pattern).
  ;; Intercepts `vertico--format-candidate' to apply per-category
  ;; text transformations BEFORE the candidate is rendered.
  ;; The &context clause ensures this method fires ONLY when
  ;; transform functions are active (set via multiform settings).
  ;; WARNING: `vertico--format-candidate' is an internal API.
  ;; Current signature (v2.11): (cand prefix suffix index _start).
  ;; ──────────────────────────────────────────────────────────────
  (defvar ar/vertico-transform-functions nil
    "Buffer-local list of transform functions applied to candidates.")

  (cl-defmethod vertico--format-candidate :around
    (cand prefix suffix index start
     &context ((not ar/vertico-transform-functions) null))
    (dolist (fun (ensure-list ar/vertico-transform-functions))
      (setq cand (funcall fun cand)))
    (cl-call-next-method cand prefix suffix index start))

  ;; ──────────────────────────────────────────────────────────────
  ;; Transform: Highlight directories with a trailing slash.
  ;; ──────────────────────────────────────────────────────────────
  (defun ar/vertico-highlight-directory-fn (file)
    "Highlight FILE as a directory if it ends with a slash."
    (if (string-suffix-p "/" file)
        (propertize file 'face 'dired-directory)
      file))

  ;; ──────────────────────────────────────────────────────────────
  ;; Transform: Highlight already-enabled modes in M-x.
  ;; Uses `minibuffer-selected-window' to robustly resolve the
  ;; buffer that was active BEFORE the minibuffer opened, instead
  ;; of the fragile `(nth 1 (buffer-list))' indexing.
  ;; ──────────────────────────────────────────────────────────────
  (defun ar/vertico-highlight-enabled-mode-fn (cmd)
    "Highlight CMD if the corresponding mode is enabled in the originating buffer."
    (let ((sym (intern-soft cmd)))
      (when sym
        (let ((origin-buf (window-buffer (minibuffer-selected-window))))
          (when (and origin-buf (buffer-live-p origin-buf))
            (with-current-buffer origin-buf
              (when (or (eq sym major-mode)
                        (and (memq sym minor-mode-list)
                             (boundp sym)
                             (symbol-value sym)))
                (add-face-text-property 0 (length cmd)
                                        'font-lock-constant-face 'append cmd)))))))
    cmd)

  ;; ──────────────────────────────────────────────────────────────
  ;; Multiform Settings: Single `setq' for the complete alist.
  ;; Commands take precedence over categories.
  ;; ──────────────────────────────────────────────────────────────
  (setq vertico-multiform-commands
        '((consult-imenu buffer)
          (consult-outline buffer)
          (consult-line-multi buffer)
          (execute-extended-command
           (ar/vertico-transform-functions . ar/vertico-highlight-enabled-mode-fn))))

  (setq vertico-multiform-categories
        '((file
           (ar/vertico-transform-functions . ar/vertico-highlight-directory-fn))
          (embark-keybinding grid)))

  ;; Activates the multiform engine post-load.
  (vertico-multiform-mode 1))

;; ──────────────────────────────────────────────────────────────
;; Remap C-s to fuzzy-search minibuffer input history.
;; Defined outside the use-package block because `consult' is
;; loaded asynchronously via Elpaca.
;; ──────────────────────────────────────────────────────────────
(defvar ar/minibuffer-maps
  '(minibuffer-local-map
    minibuffer-local-ns-map
    minibuffer-local-completion-map
    minibuffer-local-must-match-map
    minibuffer-local-isearch-map
    read-expression-map)
  "List of minibuffer keymaps.")

(with-eval-after-load 'consult
  (dolist (keymap ar/minibuffer-maps)
    (define-key (symbol-value keymap) (kbd "C-s") #'consult-history)))
#+end_src
