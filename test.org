** DONE Lexical Binding
Lexical binding is a strict requirement for modern Emacs Lisp. Unlike dynamic binding, it scopes variables to their enclosing block, enabling closures, preventing variable-bleeding bugs, and letting the byte-compiler generate faster, optimized code.

The =no-byte-compile: t= header directive stops Emacs from prematurely compiling this tangled file before =compile-angel= registers its exclusion rules.

#+begin_src emacs-lisp
;;; init.el --- Main Configuration File -*- no-byte-compile: t; lexical-binding: t; -*-
;;; Commentary:
;; This file is auto-generated by tangling `config.org`.
;; Lexical binding is enabled globally for this file via the header above.
#+end_src


** DONE Package Management
Initializes =elpaca=, an asynchronous package manager replacing =package.el=. Directory structure adheres to the =no-littering= paradigm, redirecting repositories and builds to =var/elpaca/=.

*Load-Order Physics & Asynchronous Safety:*
- =elpaca-use-package=: Installed and blocked via =(elpaca-wait)=. Ensures =:ensure= advice applies to native =use-package= prior to subsequent macro-expansion.
- =no-littering=: Installed with =:ensure (:wait t)=. Enforces XDG-like paths before asynchronously queued packages evaluate cache directories.
- =custom.el=: Deferred to =elpaca-after-init-hook=. Guarantees execution post-queue processing.
#+begin_src emacs-lisp
;; Prefer newer source files over stale byte-compiled artifacts.
(setq load-prefer-newer t)

;; Ensure use-package macro availability during byte-compilation.
(eval-when-compile (require 'use-package))
(require 'bind-key)

;; ELPACA BOOTSTRAP
(defvar elpaca-installer-version 0.12)
;; Redirect Elpaca root to var/elpaca/ (no-littering paradigm).
(defvar elpaca-directory (expand-file-name "var/elpaca/" user-emacs-directory))
(defvar elpaca-builds-directory (expand-file-name "builds/" elpaca-directory))
(defvar elpaca-sources-directory (expand-file-name "sources/" elpaca-directory))
(defvar elpaca-order '(elpaca :repo "https://github.com/progfolio/elpaca.git"
                              :ref nil :depth 1 :inherit ignore
                              :files (:defaults "elpaca-test.el" (:exclude "extensions"))
                              :build (:not elpaca-activate)))
(let* ((repo  (expand-file-name "elpaca/" elpaca-sources-directory))
       (build (expand-file-name "elpaca/" elpaca-builds-directory))
       (order (cdr elpaca-order))
       (default-directory repo))
  (add-to-list 'load-path (if (file-exists-p build) build repo))
  (unless (file-exists-p repo)
    (make-directory repo t)
    (when (<= emacs-major-version 28) (require 'subr-x))
    (condition-case-unless-debug err
        (if-let* ((buffer (pop-to-buffer-same-window "*elpaca-bootstrap*"))
                  ((zerop (apply #'call-process `("git" nil ,buffer t "clone"
                                                  ,@(when-let* ((depth (plist-get order :depth)))
                                                      (list (format "--depth=%d" depth) "--no-single-branch"))
                                                  ,(plist-get order :repo) ,repo))))
                  ((zerop (call-process "git" nil buffer t "checkout"
                                        (or (plist-get order :ref) "--"))))
                  (emacs (concat invocation-directory invocation-name))
                  ((zerop (call-process emacs nil buffer nil "-Q" "-L" "." "--batch"
                                        "--eval" "(byte-recompile-directory \".\" 0 'force)")))
                  ((require 'elpaca))
                  ((elpaca-generate-autoloads "elpaca" repo)))
            (progn (message "%s" (buffer-string)) (kill-buffer buffer))
          (error "%s" (with-current-buffer buffer (buffer-string))))
      ((error) (warn "%s" err) (delete-directory repo 'recursive))))
  (unless (require 'elpaca-autoloads nil t)
    (require 'elpaca)
    (elpaca-generate-autoloads "elpaca" repo)
    (let ((load-source-file-function nil)) (load "./elpaca-autoloads"))))
;; Exception: Internal bootstrap hook must remain `after-init-hook`.
(add-hook 'after-init-hook #'elpaca-process-queues)
(elpaca `(,@elpaca-order))

;; Use Package Integration
(elpaca elpaca-use-package
  (elpaca-use-package-mode)
  (setq elpaca-use-package-by-default t))

;; Block init reading until elpaca-use-package advice is applied.
(elpaca-wait)

;; Use Package Configuration
(setq use-package-verbose nil
      use-package-always-ensure t
      use-package-expand-minimally t
      use-package-compute-statistics nil
      use-package-enable-imenu-support t)

;; NO-LITTERING & CUSTOM FILE
;; Synchronously load no-littering to preempt async package cache paths.
(use-package no-littering
  :ensure (:wait t)
  :demand t
  :config
  (setq custom-file (no-littering-expand-etc-file-name "custom.el"))
  ;; Defer custom.el loading until Elpaca queues finish processing.
  (add-hook 'elpaca-after-init-hook (lambda () (load custom-file 'noerror)))
  (no-littering-theme-backups))

;; Register local personal libraries.
(add-to-list 'load-path (expand-file-name "lisp" user-emacs-directory))

** DONE Custom Functions
Utility functions for environment validation, performance guards, and configuration reloading. Includes childframe and icon rendering checks for UI dependencies, a buffer-size threshold to suspend expensive minor modes, an Elpaca-aware configuration reloader, and a region/symbol fallback for text operations.

#+begin_src emacs-lisp
(defun childframe-workable-p ()
  "Return non-nil if childframes are supported.
Requires a graphical display or terminal with childframe capabilities.
Utilized by UI packages such as `corfu' and `posframe'."
  (and (not noninteractive)
       (or (display-graphic-p)
           (featurep 'tty-child-frames))
       (eq (frame-parameter (selected-frame) 'minibuffer) 't)))

(defun childframe-completion-workable-p ()
  "Return non-nil if childframe completion is supported."
  (childframe-workable-p))

(defun icons-displayable-p ()
  "Return non-nil if icon fonts are displayable.
Prevents rendering attempts in environments lacking `nerd-icons' support."
  (or (featurep 'nerd-icons)
      (require 'nerd-icons nil t)))

(defun too-long-file-p ()
  "Return non-nil if the current buffer exceeds size thresholds.
Disables expensive minor modes in massive buffers to prevent main-thread freezing."
  (or (> (buffer-size) 500000)
      (and (fboundp 'buffer-line-statistics)
           (> (car (buffer-line-statistics)) 10000))))

(defun reload-init-file ()
  "Tangle and reload the Emacs configuration from `config.org'.
Processes Elpaca's asynchronous queues post-reload. Emacs cannot natively
unload packages or clear modified hooks; a full restart is required for
structural teardown."
  (interactive)
  (let* ((config-org (expand-file-name "config.org" user-emacs-directory))
         (init-el (expand-file-name "init.el" user-emacs-directory))
         (start-time (current-time))
         (divider (propertize "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━" 'face 'shadow)))

    (unless (file-exists-p config-org)
      (user-error "Configuration file %s not found" config-org))

    (when (y-or-n-p "Reload configuration? (Teardown requires restart) ")
      (message "%s" divider)
      (message "%s" (propertize "Tangling config.org..." 'face 'bold))
      (require 'org)

      (condition-case err
          (org-babel-tangle-file config-org init-el "emacs-lisp")
        (error
         (user-error "Tangle failed: %s" (error-message-string err))))

      (message "%s %s" (propertize "✓" 'face 'success) "Tangled")
      (message "%s" (propertize "Loading init.el..." 'face 'bold))

      (condition-case err
          (load-file init-el)
        (error
         (user-error "Load failed: %s" (error-message-string err))))

      ;; Process Elpaca queues to resolve newly added dependencies.
      (when (fboundp 'elpaca-process-queues)
        (message "%s" (propertize "Processing Elpaca queues..." 'face 'bold))
        (elpaca-process-queues))

      ;; Reclaim orphaned Lisp objects generated by re-evaluation.
      (garbage-collect)

      (let ((elapsed-time (float-time (time-subtract (current-time) start-time))))
        (message "%s %s" (propertize "✓" 'face 'success) "Loaded")
        (message "%s" divider)
        (message "Completed in %s seconds"
                 (propertize (format "%.2f" elapsed-time) 'face 'warning))
        (message "")
        (message "%s" (propertize "RESTART MANDATE:" 'face 'error))
        (message "  %s Additions applied: variables, packages, keybindings."
                 (propertize "•" 'face 'font-lock-builtin-face))
        (message "  %s Deletions persist: packages and hooks remain in memory."
                 (propertize "•" 'face 'font-lock-builtin-face))
        (message "  %s Removed packages persist in var/elpaca/."
                 (propertize "•" 'face 'font-lock-builtin-face))
        (message "  %s Full restart required for structural teardown."
                 (propertize "•" 'face 'font-lock-builtin-face))
        (message "%s" divider)))))

(defun selected-region-or-symbol-at-point ()
  "Return the active region, or the symbol at point if inactive.
Provides a fallback for text operations lacking an explicit region."
  (if (region-active-p)
      (buffer-substring-no-properties (region-beginning) (region-end))
    (thing-at-point 'symbol t)))

** DONE Garbage Collector
Manages garbage collection via =gcmh= idle timers. Elevates the GC threshold during active input and triggers background collection during idle periods. Disables =gcmh= on experimental MPS Incremental GC (=igc=) builds and utilizes =after-focus-change-function= for focus-out memory reclamation.

#+begin_src emacs-lisp
(use-package gcmh
  ;; Disabled on experimental MPS Incremental GC (igc) builds.
  ;; gcmh targets the legacy stop-the-world GC and conflicts with igc.
  :unless (fboundp 'igc-info)
  :hook (elpaca-after-init . gcmh-mode)
  :init
  (setq gcmh-idle-delay 'auto
        gcmh-auto-idle-delay-factor 10
        ;; 128MB baseline threshold to prevent LSP-induced GC stutters.
        gcmh-high-cons-threshold (* 128 1024 1024)))

;; Triggers GC on frame focus loss. Replaces obsolete `focus-out-hook`.
(add-function :after after-focus-change-function
              (lambda ()
                (unless (frame-focus-state)
                  (garbage-collect))))
#+end_src

** DONE Delete Pair
Configures the native =delete-pair= engine and provides an O(1) DWIM wrapper for adjacent, string, and nested pair deletion, strictly confined to programming modes.
#+begin_src emacs-lisp

(setq delete-pair-blink-delay 0.1)

;; Emacs 31+: Push mark before deletion for inner-text recovery.

(defun ar/delete-pair-dwim ()
  "Intelligently delete the pair surrounding or preceding the cursor.
Strictly confined to `prog-mode' to prevent structural destruction in prose."
  (interactive)
  (unless (derived-mode-p 'prog-mode)
    (user-error "Delete Pair DWIM restricted to prog-mode; use `ds` in prose"))
  (let ((prev (char-before))
        (next (char-after))
        (ppss (syntax-ppss)))
    (cond
     ;; Delete preceding closing delimiter or quote.
     ((and prev (memq (char-syntax prev) '(?\) ?\" ?\|)))
      (delete-pair -1))
     ;; Delete following opening delimiter or quote.
     ((and next (memq (char-syntax next) '(?\( ?\" ?\|)))
      (delete-pair 1))
     ;; Inside a string: jump to start, handle prefixes, and delete.
     ((nth 3 ppss)
      (let ((start (nth 8 ppss)))
        (goto-char start)
        ;; Handle string prefixes (e.g., f"..." in Python).
        (unless (memq (char-after) '(?\" ?\|))
          (forward-char 1))
        (delete-pair 1)))
     ;; Fallback: jump out of current list and delete enclosing pair.
     (t
      (condition-case nil
          (progn
            (backward-up-list)
            (delete-pair 1))
        (scan-error (user-error "Not inside a deletable pair")))))))

(use-package orderless
  :custom
  ;; Global completion styles with =basic= as a dynamic fallback.
  (completion-styles '(orderless basic))
  ;; Disables default category overrides to prevent external style injection.
  (completion-category-defaults nil)
  ;; Emacs 31+: Enables substring-like behavior for partial-completion.
  (completion-pcm-leading-wildcard t)
  ;; Strict global matching styles, omitting flex and initialism for performance.
  (orderless-matching-styles '(orderless-regexp orderless-literal))
  :config
  ;; Unified category routing alist for files and LSP payloads.
  (setq completion-category-overrides
        `((file (styles basic partial-completion orderless))
          (eglot (styles ,@completion-styles))
          (eglot-capf (styles ,@completion-styles))))
  ;; Explicit affix dictionary for predictable Vim-like dispatch routing.
  (setq orderless-affix-dispatch-alist
        `((?! . orderless-without-literal)
          (?= . orderless-literal)
          (?% . orderless-initialism)
          (?~ . orderless-flex)
          (?, . orderless-literal-prefix)
          (?& . ,(if (fboundp 'orderless-annotation)
                     'orderless-annotation
                   'orderless-literal))))
  ;; Handles Consult disambiguation suffixes and file extension matching.
  (defun ar/orderless-disambiguation-dispatch (word _index _total)
    "Handle disambiguation suffixes for Consult and file extensions."
    (let ((tofu-re (if (boundp 'consult--tofu-regexp)
                       (concat consult--tofu-regexp "*\\'")
                     "\\'")))
      (cond
       ;; Exact end-of-string match for Consult commands.
       ((string-suffix-p "$" word)
        `(orderless-regexp . ,(concat (substring word 0 -1) tofu-re)))
       ;; File extension matching in minibuffer and shell modes.
       ((and (or minibuffer-completing-file-name
                 (derived-mode-p 'eshell-mode))
             (string-prefix-p "." word))
        `(orderless-regexp . ,(concat "\\." (substring word 1) tofu-re))))))
  ;; Chains the native affix dispatcher with the custom disambiguation dispatcher.
  (setq orderless-style-dispatchers
        '(orderless-affix-dispatch
          ar/orderless-disambiguation-dispatch)))

(defun ar/capture-url-from-clipboard ()
  "Return the URL currently in the clipboard/kill-ring.
Falls back to `read-string' if no HTTP/HTTPS URL is found."
  ;; `current-kill' natively forces an interprogram paste query on PGTK/Wayland
  ;; before falling back, preventing stale `(car kill-ring)' data.
  (let ((clip (string-trim (or (ignore-errors (current-kill 0)) ""))))
    (if (string-match-p "^https?://" clip)
        clip
      (read-string "URL: "))))

(defun ar/capture-url-as-org-link ()
  "Format the clipboard URL as an org link with a prompted description."
  (let* ((url  (ar/capture-url-from-clipboard))
         (desc (read-string (format "Description [%s]: " url) nil nil url)))
    (format "[[%s][%s]]" url desc)))

(defun ar/capture-habit-range-schedule ()
  "Prompt for ideal and max repeat intervals, return a SCHEDULED timestamp.
Uses `read-string' because %^{prompt} inside %(sexp) receives the literal
string, not user input, per the Org manual expansion order."
  (let* ((ideal   (read-string "Ideal interval in days (e.g. 1): " "1"))
         (ideal-n (string-to-number ideal))
         (max-def (number-to-string (* 2 ideal-n)))
         (max     (read-string
                   (format "Max interval in days (default %s): " max-def)
                   max-def)))
    (format-time-string
     (format "<%%Y-%%m-%%d %%a .+%sd/%sd>" ideal max))))

(defun ar/capture-dispatch ()
  "Hierarchical smart capture dispatcher.
Top-level keys:
[t] Task → inbox in todo.org    [n] New Denote note
[j] Journal (sub-menu)          [m] Meeting notes
[b] Book / article              [h] Habit (sub-menu)
[g] Goal (sub-menu)             [w] Web clip (sub-menu)
[i] Quick inbox with context    [q] Quit"
  (interactive)
  (let ((choice (read-char-choice
                 (concat "Capture: "
                         "[t]ask [n]ote [j]ournal [m]eeting [b]ook "
                         "[h]abit [g]oal [w]eb [i]nbox [q]uit: ")
                 '(?t ?n ?j ?m ?b ?h ?g ?w ?i ?q))))
    (pcase choice
      (?t (org-capture nil "i"))
      (?n (call-interactively #'denote))
      (?j (let ((kind (read-char-choice
                       "Journal: [d]aily  [l]ink+context  [q]uit: "
                       '(?d ?l ?q))))
            (pcase kind
              (?d (org-capture nil "jd"))
              (?l (org-capture nil "jl"))
              (?q (keyboard-quit)))))
      (?m (org-capture nil "m"))
      (?b (org-capture nil "b"))
      (?h (let ((kind (read-char-choice
                       "Habit: [d]aily flexible  [s]trict daily  [r]ange  [q]uit: "
                       '(?d ?s ?r ?q))))
            (pcase kind
              (?d (org-capture nil "hd"))
              (?s (org-capture nil "hs"))
              (?r (org-capture nil "hr"))
              (?q (keyboard-quit)))))
      (?g (let ((kind (read-char-choice
                       "Goal: [l]ife  [Q]uarterly  [w]eekly  [q]uit: "
                       '(?l ?Q ?w ?q))))
            (pcase kind
              (?l (org-capture nil "gl"))
              (?Q (org-capture nil "gQ"))
              (?w (org-capture nil "gw"))
              (?q (keyboard-quit)))))
      (?w (let ((kind (read-char-choice
                       "Web: [c]lipboard URL  [l]ink+context  [q]uit: "
                       '(?c ?l ?q))))
            (pcase kind
              (?c (org-capture nil "wc"))
              (?l (org-capture nil "wl"))
              (?q (keyboard-quit)))))
      (?i (org-capture nil "i"))
      (?q (keyboard-quit)))))

** DONE Org Structure Templates
Configures Org structure templates for rapid block insertion and implements a deferred loading engine for Babel language backends.
#+begin_src emacs-lisp
(with-eval-after-load 'org
  ;; Structure Templates (org-tempo)
  (require 'org-tempo)
  (dolist (pair '(("sh" . "src shell")
                  ("el" . "src emacs-lisp")
                  ("py" . "src python")
                  ("dot" . "src dot")
                  ("plantuml" . "src plantuml")
                  ("tex" . "src latex")
                  ("yml" . "src yaml")
                  ("kbd" . "src kbd")
                  ("kdl" . "src kdl")))
    (add-to-list 'org-structure-template-alist pair))

  ;; Babel Lazy-Loading Engine
  ;; NOTE: Jupyter/Python Paradigm Shift Deferred
  ;; `emacs-jupyter` relies on heavy, stateful ZeroMQ kernel initialization hooks
  ;; that expect `ob-jupyter` to be fully loaded BEFORE interacting with a block.
  ;; The lazy-loader only triggers on execution (C-c C-c), which is mathematically
  ;; too late for the kernel handshake. To implement `emacs-jupyter` in the future,
  ;; explicitly `(require 'ob-jupyter)` in your Jupyter setup block, add
  ;; `(python . jupyter)` to `ar/org-babel-mode-alist`, and map `"jpy"` to
  ;; `"src jupyter-python"` in the structure templates above. For now, standard
  ;; `ob-python` is utilized.

  (defvar ar/org-babel-mode-alist
    '((c . C) (cpp . C) (C++ . C) (D . C)
      (elisp . emacs-lisp) (sh . shell) (bash . shell)
      (matlab . octave) (rust . rustic-babel) (amm . ammonite))
    "Alist mapping divergent block headers to their `ob-*` package names.")

  (defun ar/org--babel-lazy-load (lang)
    "Lazy load the babel package for LANG and register it."
    (cl-check-type lang (or symbol null))
    ;; Guard against nil languages to prevent `require` crashes during export.
    (when lang
      (unless (cdr (assq lang org-babel-load-languages))
        (prog1 (or (require (intern (format "ob-%s" lang)) nil t)
                   (require lang nil t))
          ;; Update the registry so Org's execution engine recognizes it.
          (add-to-list 'org-babel-load-languages (cons lang t))))))

  ;; 1. EXPORT HOOK: Lazy load when exporting the document.
  (define-advice org-babel-exp-src-block (:before (&optional element) lazy-load-export)
    "Lazy load a babel package when a block is processed during exporting."
    (let* ((info (org-babel-get-src-block-info nil element))
           (lang (nth 0 info))
           (lang (cond ((symbolp lang) lang) ((stringp lang) (intern lang))))
           (lang (or (cdr (assq lang ar/org-babel-mode-alist)) lang)))
      (ar/org--babel-lazy-load lang)))

  ;; 2. SYNTAX HOOK: Lazy load when opening a block edit buffer (C-c ').
  (define-advice org-src--get-lang-mode (:before (lang) lazy-load-syntax)
    "Lazy load a babel package to ensure syntax highlighting in edit buffers."
    (let* ((lang-str (if (stringp lang) lang (symbol-name lang)))
           (lang-sym (intern lang-str)))
      ;; Use `assoc-string` to correctly match string keys in `org-src-lang-modes`.
      (unless (cdr (assoc-string lang-str org-src-lang-modes))
        (ar/org--babel-lazy-load lang-sym))))

  ;; 3. EXECUTION HOOK: Lazy load when executing the block (C-c C-c).
  ;; Uses `:after-while` to ensure the package only loads if the user
  ;; actually confirms execution (or if confirmation is disabled).
  (define-advice org-babel-confirm-evaluate (:after-while (info) lazy-load-execute)
    "Load babel libraries lazily when babel blocks are executed."
    (let* ((lang (nth 0 info))
           (lang (cond ((symbolp lang) lang) ((stringp lang) (intern lang))))
           (lang (or (cdr (assq lang ar/org-babel-mode-alist)) lang)))
      (ar/org--babel-lazy-load lang)
      t))

  ;; 4. THE OVERRIDE: Neuter the synchronous startup tax.
  (advice-add #'org-babel-do-load-languages :override #'ignore))

             (dir (file-name-directory new-name)))
        (unless (file-directory-p dir)
          (make-directory dir t))
        (ignore-errors
          (let ((edits (eglot--request
                        server
                        :workspace/willRenameFiles
                        `(:files [(:oldUri ,old-uri :newUri ,new-uri)]))))
            (when edits
              (eglot--apply-workspace-edit server edits 'ar/eglot-rename-file))))
        (rename-file old-name new-name 1)
        (set-visited-file-name new-name nil t)
        (jsonrpc-notify server :workspace/didRenameFiles
                        `(:files [(:oldUri ,old-uri :newUri ,new-uri)]))
        (message "Renamed %s → %s and updated LSP references"
                 (file-name-nondirectory old-name)
                 (file-name-nondirectory new-name)))))

    "Display the LSP moniker for the symbol at point in the echo area."

          (user-error "No active Eglot server")
        (if (not (eglot-server-capable :monikerProvider))
